# --------------------------------------------------------------------------- #
# Public exception
# --------------------------------------------------------------------------- #
class TransientError(Exception):
    """Raised for retryable network errors."""


# --------------------------------------------------------------------------- #
# HTTP transport
# --------------------------------------------------------------------------- #
def fetch_http_page(url: str, cursor: Optional[str]) -> Dict[str, Any]:
    """
    Perform a GET request to *url*, optionally adding a `cursor` query
    parameter.  The response is expected to be UTF‑8 encoded JSON.
    HTTP 429, 5xx and urllib.error.URLError are considered transient.
    """
    # Build URL with cursor if present
    parsed = urllib.parse.urlparse(url)
    query = urllib.parse.parse_qs(parsed.query, keep_blank_values=True)
    if cursor is not None:
        query["cursor"] = [cursor]
    new_query = urllib.parse.urlencode(query, doseq=True)
    new_url = urllib.parse.urlunparse(
        (parsed.scheme, parsed.netloc, parsed.path, parsed.params, new_query, parsed.fragment)
    )

    try:
        with urllib.request.urlopen(new_url) as resp:
            data = resp.read().decode("utf-8")
            return json.loads(data)
    except urllib.error.HTTPError as e:
        # Retryable status codes
        if e.code == 429 or 500 <= e.code < 600:
            raise TransientError(f"HTTP {e.code}") from e
        raise
    except urllib.error.URLError as e:
        raise TransientError("Network error") from e
