# API-to-SQLite synchronization specification

## Contract

Implement `sync_app.py` using Python's standard library. Expose `sync(connection, fetch_page, sleep, *, max_attempts=3, base_delay=0.1, logger=None) -> int`, `fetch_http_page(url, cursor)`, and `main(argv=None) -> int`. The model may add helpers but must preserve these signatures. `connection` is a `sqlite3.Connection`; `fetch_page(cursor)` returns a decoded page dictionary. `sleep(seconds)` is injectable. The sync result is the number of unique records committed.

Each page has exactly the required keys `items` (list) and `next_cursor` (null or a nonempty string). An item is a dictionary with `id` (nonempty string), `name` (nonempty string), and `updated_at` (nonempty ISO 8601 timestamp including a timezone). Preserve the timestamp text. Extra page or item keys are permitted and ignored. Duplicate IDs across pages are resolved by the last occurrence in fetch order.

Fetch the first page using cursor `None`, then follow `next_cursor` until null. Detect a repeated cursor and raise `ValueError`. Invalid pages or records raise `ValueError`. All pages must be validated before any persistent database change. Upsert into `records(id TEXT PRIMARY KEY, name TEXT NOT NULL, updated_at TEXT NOT NULL)` in one transaction; repeated runs must not duplicate rows. A failed fetch, validation, or database operation leaves the database as it was before the call, including if the table did not previously exist. Do not commit or roll back a transaction already owned by the caller: reject an already-active transaction with `ValueError` before fetching.

Define `TransientError` for retryable fetch failures. Retry each individual page at most `max_attempts` total calls, sleeping `base_delay * 2**(attempt-1)` after each non-final `TransientError`. Other exceptions propagate immediately. Validate `max_attempts` is a positive integer excluding booleans and `base_delay` is a nonnegative finite number before fetching. Log successful completion without logging credentials or record contents.

`fetch_http_page` performs a GET using `urllib.request`, appending a URL-encoded `cursor` query parameter only when cursor is not null, preserving existing query parameters. It decodes UTF-8 JSON. HTTP 429, HTTP 5xx, and `urllib.error.URLError` raise `TransientError`; other HTTP errors propagate. No network access is needed in tests.

# Transient errors are retryable (429, 5xx, network errors)
TRANSIENT_STATUS_CODES = {429} | set(str(code) for code in range(500, 600))


class TransientError(Exception):
    """Raised for transient failures that should be retried."""
    pass


def validate_config(max_attempts: int, base_delay: float) -> None:
    """Validate retry configuration parameters."""
    if isinstance(max_attempts, bool):
        raise ValueError("max_attempts must be a positive integer, not a boolean")
    if not isinstance(max_attempts, int) or max_attempts <= 0:
        raise ValueError("max_attempts must be a positive integer")
    
    if isinstance(base_delay, bool):
        raise ValueError("base_delay must be a nonnegative finite number, not a boolean")
    if not isinstance(base_delay, (int, float)):
        raise ValueError("base_delay must be a nonnegative finite number")
    if base_delay < 0:
        raise ValueError("base_delay must be a nonnegative finite number")
    if not (isinstance(base_delay, float) and base_delay != float('inf') and base_delay != float('-inf')):
        # Check for NaN specifically using math.isfinite or try/except
        import math
        if not math.isfinite(base_delay):
            raise ValueError("base_delay must be a nonnegative finite number")


def fetch_http_page(url: str, cursor: str) -> dict:
    """Fetch a page from the API."""
    base_url = url.split("?")[0]
    params = parse_qs(urlsplit(url).query)
    
    if cursor is not None and cursor != "":
        params["cursor"] = [cursor]
    
    query_string = "&".join(f"{k}={v}" for k, v in params.items())
    full_url = f"{base_url}?{query_string}" if query_string else base_url
    
    try:
        with urllib.request.urlopen(full_url) as response:
            data = response.read().decode("utf-8")
            return json.loads(data)
    except HTTPError as e:
        if e.code in TRANSIENT_STATUS_CODES:
            raise TransientError(f"Transient error {e.code}: {e.reason}") from e
        raise
    
    except URLError as e:
        raise TransientError(f"Network error: {e.reason}") from e


def validate_page(page: dict) -> None:
    """Validate a page structure."""
    if not isinstance(page, dict):
        raise ValueError("Page must be a dictionary")
    
    if "items" not in page or not isinstance(page["items"], list):
        raise ValueError("Page must contain 'items' as a list")
    
    if "next_cursor" not in page:
        raise ValueError("Page must contain 'next_cursor'")
    
    next_cursor = page["next_cursor"]
    if next_cursor is not None and not isinstance(next_cursor, str):
        raise ValueError("'next_cursor' must be a string or null")


def validate_item(item: dict) -> None:
    """Validate an item structure."""
    if not isinstance(item, dict):
        raise ValueError("Item must be a dictionary")
    
    required_keys = {"id", "name", "updated_at"}
    missing = required_keys - set(item.keys())
    if missing:
        raise ValueError(f"Item missing required keys: {missing}")
    
    for key in required_keys:
        val = item[key]
        if not isinstance(val, str) or val == "":
            raise ValueError(f"Field '{key}' must be a nonempty string")


def validate_records(records: list) -> None:
    """Validate all records in a page."""
    for i, record in enumerate(records):
        try:
            validate_item(record)
        except ValueError as e:
            raise ValueError(f"Invalid record at index {i}: {e}") from e


def sync(connection: sqlite3.Connection, fetch_page, sleep, *, max_attempts: int = 3, base_delay: float = 0.1, logger=None) -> int:
    """Synchronize data from API to SQLite."""
    validate_config(max_attempts, base_delay)
    
    # Check for active transaction
    try:
        connection.execute("PRAGMA busy_timeout=5000")
    except sqlite3.OperationalError:
        raise ValueError("An active transaction is already in progress") from None
    
    # Fetch all pages first
    cursor = None
    pages = []
    
    while True:
        try:
            page = fetch_page(cursor)
        except TransientError as e:
            if logger:
                logger.warning(f"Transient error fetching page {cursor}: {e}")
            raise
        
        validate_page(page)
        
        # Check for cycle
        if cursor is not None and page["next_cursor"] == cursor:
            raise ValueError("Repeated cursor detected")
        
        pages.append(page)
        
        next_cursor = page.get("next_cursor")
        if next_cursor is None or next_cursor == "":
            break
        
        # Check for cycle before moving to next
        if next_cursor in [p["next_cursor"] for p in pages[:-1]]:
            raise ValueError("Repeated cursor detected")
        
        cursor = next_cursor
    
    # Validate all records across all pages
    all_records = []
    for page in pages:
        validate_page(page)
        validate_records(page["items"])
        all_records.extend(page["items"])
    
    if not all_records:
        return 0
    
    # Upsert into database
    try:
        connection.execute("BEGIN")
        
        for record in all_records:
            conn = connection
            conn.execute(
                "INSERT OR REPLACE INTO records (id, name, updated_at) VALUES (?, ?, ?)",
                (record["id"], record["name"], record["updated_at"])
            )
        
        connection.commit()
        return len(all_records)
    
    except Exception:
        # Rollback on any error
        try:
            connection.rollback()
        except sqlite3.Error:
            pass
        raise
