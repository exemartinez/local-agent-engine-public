import sqlite3
import json
import urllib.request
import urllib.parse
import os
import sys
import logging
from urllib.error import HTTPError, URLError


class TransientError(Exception):
    pass


def fetch_http_page(url, cursor):
    if cursor is not None:
        parsed = urllib.parse.urlparse(url)
        query_params = urllib.parse.parse_qs(parsed.query)
        query_params['cursor'] = [cursor]
        new_query = urllib.parse.urlencode(query_params, doseq=True)
        new_url = parsed._replace(query=new_query).geturl()
    else:
        new_url = url
    
    try:
        response = urllib.request.urlopen(new_url)
        data = response.read()
        return json.loads(data.decode('utf-8'))
    except HTTPError as e:
        if e.code in (429, 500, 502, 503, 504):
            raise TransientError(f"HTTP {e.code}: {e.reason}") from e
        raise
    except URLError as e:
        raise TransientError(f"Network error: {e.reason}") from e


def sync(connection, fetch_page, sleep, *, max_attempts=3, base_delay=0.1, logger=None):
    # Validate retry parameters
    if not isinstance(max_attempts, int) or max_attempts <= 0 or isinstance(max_attempts, bool):
        raise ValueError("max_attempts must be a positive integer")
    if not isinstance(base_delay, (int, float)) or base_delay < 0 or not (base_delay == base_delay):  # Check for NaN
        raise ValueError("base_delay must be a non-negative finite number")
    
    # Check for active transaction
    if connection.in_transaction:
        raise ValueError("Database connection is already in a transaction")
    
    # Fetch all pages
    pages = []
    cursor = None
    seen_cursors = set()
    
    while True:
        if cursor in seen_cursors:
            raise ValueError("Repeated cursor detected")
        seen_cursors.add(cursor)
        
        # Retry logic for each page
        attempt = 0
        while attempt < max_attempts:
            try:
                page = fetch_page(cursor)
                break
            except TransientError as e:
                attempt += 1
                if attempt >= max_attempts:
                    raise
                sleep(base_delay * (2 ** (attempt - 1)))
        
        # Validate page structure
        if not isinstance(page, dict):
            raise ValueError("Page must be a dictionary")
        if "items" not in page or "next_cursor" not in page:
            raise ValueError("Page must have 'items' and 'next_cursor' keys")
        if not isinstance(page["items"], list):
            raise ValueError("Page 'items' must be a list")
        if page["next_cursor"] is not None and not isinstance(page["next_cursor"], str):
            raise ValueError("Page 'next_cursor' must be None or a non-empty string")
        
        pages.append(page)
        
        # Check if we're done
        if page["next_cursor"] is None:
            break
        cursor = page["next_cursor"]
    
    # Validate all items
    all_items = []
    for page in pages:
        for item in page["items"]:
            if not isinstance(item, dict):
                raise ValueError("Item must be a dictionary")
            if "id" not in item or "name" not in item or "updated_at" not in item:
                raise ValueError("Item must have 'id', 'name', and 'updated_at'")
            if not isinstance(item["id"], str) or not item["id"]:
                raise ValueError("Item 'id' must be a non-empty string")
            if not isinstance(item["name"], str) or not item["name"]:
                raise ValueError("Item 'name' must be a non-empty string")
            if not isinstance(item["updated_at"], str) or not item["updated_at"]:
                raise ValueError("Item 'updated_at' must be a non-empty string")
            all_items.append(item)
    
    # Deduplicate items by ID (last wins)
    unique_items = {}
    for item in all_items:
        unique_items[item["id"]] = item
    
    # Prepare data for database
    records = [(item["id"], item["name"], item["updated_at"]) for item in unique_items.values()]
    
    # Perform atomic database operation
    try:
        with connection:
            # Create table if needed
            connection.execute("""
                CREATE TABLE IF NOT EXISTS records (
                    id TEXT PRIMARY KEY,
                    name TEXT NOT NULL,
                    updated_at TEXT NOT NULL
                )
            """)
            
            # Upsert records
            connection.executemany("""
                INSERT OR REPLACE INTO records (id, name, updated_at)
                VALUES (?, ?, ?)
            """, records)
            
            if logger:
                logger.info(f"Synced {len(records)} unique records")
            
            return len(records)
    except Exception:
        # Transaction is automatically rolled back if any error occurs
        raise


def main(argv=None):
    if argv is None:
        argv = sys.argv[1:]
    
    # Parse arguments
    args = {}
    i = 0
    while i < len(argv):
        if argv[i] == "--url":
            args["url"] = argv[i + 1]
            i += 2
        elif argv[i] == "--db":
            args["db"] = argv[i + 1]
            i += 2
        elif argv[i] == "--max-attempts":
            args["max_attempts"] = argv[i + 1]
            i += 2
        elif argv[i] == "--base-delay":
            args["base_delay"] = argv[i + 1]
            i += 2
        else:
            print(f"Unknown argument: {argv[i]}", file=sys.stderr)
            return 2
        i += 1
    
    # Get environment variables
    url = args.get("url") or os.environ.get("SYNC_URL")
    db_path = args.get("db") or os.environ.get("SYNC_DB")
    max_attempts = args.get("max_attempts") or os.environ.get("SYNC_MAX_ATTEMPTS")
    base_delay = args.get("base_delay") or os.environ.get("SYNC_BASE_DELAY")
    
    # Validate required parameters
    if not url or not db_path:
        print("Missing required --url or --db, or missing SYNC_URL or SYNC_DB environment variables", file=sys.stderr)
        return 2
    
    # Parse retry parameters
    try:
        if max_attempts is not None:
            max_attempts = int(max_attempts)
        else:
            max_attempts = 3
        if base_delay is not None:
            base_delay = float(base_delay)
        else:
            base_delay = 0.1
    except (ValueError, TypeError):
        print("Invalid max-attempts or base-delay value", file=sys.stderr)
        return 2
    
    # Setup logging
    logger = logging.getLogger("sync_app")
    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler(sys.stderr)
    handler.setFormatter(logging.Formatter('%(asctime)s - %(levelname)s - %(message)s'))
    logger.addHandler(handler)
    
    # Create database connection
    try:
        connection = sqlite3.connect(db_path)
        connection.execute("PRAGMA foreign_keys = ON")
    except Exception as e:
        print(f"Failed to connect to database: {e}", file=sys.stderr)
        return 1
    
    # Fetch page function
    def fetch_page(cursor):
        return fetch_http_page(url, cursor)
    
    # Sleep function
    import time
    sleep = time.sleep
    
    try:
        result = sync(connection, fetch_page, sleep, max_attempts=max_attempts, base_delay=base_delay, logger=logger)
        connection.close()
        return 0
    except Exception as e:
        connection.close()
        print(f"Sync failed: {e}", file=sys.stderr)
        return 1
