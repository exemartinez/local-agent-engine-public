## Contract

Expose `sync(connection, fetch_page, sleep, *, max_attempts=3, base_delay=0.1, logger=None) -> int`, `fetch_http_page(url, cursor)`, and `main(argv=None) -> int`. The model may add helpers but must preserve these signatures. `connection` is a `sqlite3.Connection`; `fetch_page(cursor)` returns a decoded page dictionary. `sleep(seconds)` is injectable. The sync result is the number of unique records committed.

Each page has exactly the required keys `items` (list) and `next_cursor` (null or a nonempty string). An item is a dictionary with `id` (nonempty string), `name` (nonempty string), and `updated_at` (nonempty ISO 8601 timestamp including a timezone). Preserve the timestamp text. Extra page or item keys are permitted and ignored. Duplicate IDs across pages are resolved by the last occurrence in fetch order.

Fetch the first page using cursor `None`, then follow `next_cursor` until null. Detect a repeated cursor and raise `ValueError`. Invalid pages or records raise `ValueError`. A failed fetch, validation, or database operation leaves the database as it was before the call, including if the table did not previously exist. Do not commit or roll back a transaction already owned by the caller: reject an already-active transaction with `ValueError` before fetching.

Define `TransientError` for retryable fetch failures. Retry each individual page at most `max_attempts` total calls, sleeping `base_delay * 2**(attempt-1)` after each non-final `TransientError`. Other exceptions propagate immediately. Validate `max_attempts` is a positive integer excluding booleans and `base_delay` is a nonnegative finite number before fetching. from urllib.error import HTTPError, URLError

class TransientError(Exception):
    pass

def fetch_http_page(url, cursor):
    params = {}
    if cursor is not None:
        params['cursor'] = cursor
    
    parsed_url = urllib.parse.urlparse(url)
    query = urllib.parse.parse_qs(parsed_url.query)
    query.update(params)
    
    # Reconstruct URL with new query parameters
    new_query = urllib.parse.urlencode(query, doseq=True)
    new_url = urllib.parse.urlunparse((
        parsed_url.scheme,
        parsed_url.netloc,
        parsed_url.path,
        parsed_url.params,
        new_query,
        parsed_url.fragment
    ))
    
    try:
        response = urllib.request.urlopen(new_url)
        data = response.read().decode('utf-8')
        return json.loads(data)
    except HTTPError as e:
        if e.code in (429, 500, 502, 503):
            raise TransientError() from e
        else:
            raise
    except URLError:
        raise TransientError()

def sync(connection, fetch_page, sleep, *, max_attempts=3, base_delay=0.1, logger=None):
    # Validate inputs
    if not isinstance(max_attempts, int) or max_attempts <= 0 or isinstance(max_attempts, bool):
        raise ValueError("max_attempts must be a positive integer")
    
    if not isinstance(base_delay, (int, float)) or base_delay < 0 or base_delay != base_delay:  # Check for NaN
        raise ValueError("base_delay must be a non-negative finite number")
    
    # Check if transaction is already active
    if connection.in_transaction:
        raise ValueError("Database connection is already in a transaction")
    
    # Fetch all pages
    pages = []
    current_cursor = None
    seen_cursors = set()
    
    for attempt in range(max_attempts):
        try:
            page = fetch_page(current_cursor)
            break
        except TransientError:
            if attempt == max_attempts - 1:
                raise
            sleep(base_delay * (2 ** attempt))
    else:
        # This shouldn't be reached due to the loop structure, but just in case
        raise TransientError()
    
    while True:
        # Validate page structure
        if not isinstance(page, dict):
            raise ValueError("Invalid page: must be a dictionary")
        
        if 'items' not in page or 'next_cursor' not in page:
            raise ValueError("Invalid page: missing required keys")
        
        items = page['items']
        next_cursor = page['next_cursor']
        
        if not isinstance(items, list):
            raise ValueError("Invalid page: items must be a list")
        
        # Validate each item
        for item in items:
            if not isinstance(item, dict):
                raise ValueError("Invalid item: must be a dictionary")
            
            required_keys = ['id', 'name', 'updated_at']
            for key in required_keys:
                if key not in item or not isinstance(item[key], str) or not item[key]:
                    raise ValueError(f"Invalid item: missing or empty {key}")
            
            # Validate timestamp format (basic check)
            timestamp = item['updated_at']
            if not timestamp.endswith(('+00:00', 'Z')) and not any(tz in timestamp for tz in ['+', '-', 'T']):
                raise ValueError("Invalid timestamp format")
        
        pages.append(page)
        
        # Check for repeated cursor
        if next_cursor in seen_cursors:
            raise ValueError("Repeated cursor detected")
        seen_cursors.add(next_cursor)
        
        # Check if we're done
        if next_cursor is None:
            break
        
        # Fetch next page with retries
        current_cursor = next_cursor
        for attempt in range(max_attempts):
            try:
                page = fetch_page(current_cursor)
                break
            except TransientError:
                if attempt == max_attempts - 1:
                    raise
                sleep(base_delay * (2 ** attempt))
        else:
            # This shouldn't be reached due to the loop structure, but just in case
            raise TransientError()
    
    # Prepare data for upsert
    records = {}
    for page in pages:
        for item in page['items']:
            records[item['id']] = (item['name'], item['updated_at'])
    
    # Perform database operation in a transaction
    try:
        with connection:
            # Create table if it doesn't exist
            connection.execute("""
                CREATE TABLE IF NOT EXISTS records (
                    id TEXT PRIMARY KEY,
                    name TEXT NOT NULL,
                    updated_at TEXT NOT NULL
                )
            """)
            
            # Upsert records
            upsert_sql = """
                INSERT OR REPLACE INTO records (id, name, updated_at)
                VALUES (?, ?, ?)
            """
            
            for record_id, (name, updated_at) in records.items():
                connection.execute(upsert_sql, (record_id, name, updated_at))
            
            # Return count of unique records
            count = len(records)
            
            if logger:
                logger.info(f"Sync completed successfully, committed {count} unique records")
                
            return count
            
    except Exception as e:
        # If we get here, the transaction was rolled back automatically
        raise
