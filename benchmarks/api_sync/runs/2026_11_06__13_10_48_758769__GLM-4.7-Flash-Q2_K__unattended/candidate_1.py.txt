class TransientError(Exception):

def fetch_http_page(url: str, cursor: Optional[str]) -> Dict:
    """
    Performs a GET request using urllib.request.
    Appends URL-encoded cursor query parameter only when cursor is not null.
    Decodes UTF-8 JSON.
    HTTP 429, HTTP 5xx, and urllib.error.URLError raise TransientError.
    Other HTTP errors propagate.
    """
    try:
        # Construct URL with cursor query parameter if cursor is not None
        if cursor is not None:
            url = urllib.parse.urljoin(url, f"?cursor={urllib.parse.quote_plus(cursor)}")
        
        with urllib.request.urlopen(url) as response:
            # Check for transient errors
            status = response.getcode()
            if status in (429, 500, 502, 503):
                raise TransientError(f"Transient HTTP error: {status}")
            
            # Decode UTF-8 JSON
            data = response.read().decode('utf-8')
            return json.loads(data)
    except urllib.error.URLError as e:
        raise TransientError(f"Transient network error: {e}")
    except json.JSONDecodeError as e:
        raise ValueError(f"Invalid JSON: {e}")
