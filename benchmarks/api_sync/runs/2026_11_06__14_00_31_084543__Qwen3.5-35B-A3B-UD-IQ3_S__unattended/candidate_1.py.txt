"""API-to-SQLite synchronization module."""

import argparse
import json
import logging
import os
import sys
import time
import urllib.request
import urllib.parse
import urllib.error
import sqlite3
from typing import Optional, Dict, Any, List, Callable


class TransientError(Exception):
    """Raised for transient network errors that should be retried."""
    pass


def fetch_http_page(url: str, cursor: Optional[str]) -> Dict[str, Any]:
    """Fetch a page from the API with optional cursor parameter."""
    # Build the URL with cursor parameter if provided
    if cursor is not None:
        # Parse existing URL to preserve query parameters
        parsed = urllib.parse.urlparse(url)
        query_params = urllib.parse.parse_qs(parsed.query, keep_blank_values=True)
        query_params['cursor'] = [cursor]
        new_query = urllib.parse.urlencode(query_params, doseq=True)
        new_url = urllib.parse.urlunparse(parsed._replace(query=new_query))
    else:
        new_url = url
    
    # Make the request
    request = urllib.request.Request(new_url, method='GET')
    
    try:
        with urllib.request.urlopen(request) as response:
            data = response.read().decode('utf-8')
            return json.loads(data)
    except urllib.error.HTTPError as e:
        # HTTP 429, 5xx are transient
        if e.code == 429 or e.code >= 500:
            raise TransientError(f"HTTP {e.code}")
        raise
    except urllib.error.URLError:
        raise TransientError("Network error")


def _validate_page(page: Dict[str, Any]) -> None:
    """Validate that a page has the required structure."""
    if not isinstance(page, dict):
        raise ValueError("Page must be a dictionary")
    
    if 'items' not in page:
        raise ValueError("Page must have 'items' key")
    
    if not isinstance(page['items'], list):
        raise ValueError("'items' must be a list")
    
    if 'next_cursor' not in page:
        raise ValueError("Page must have 'next_cursor' key")
    
    next_cursor = page['next_cursor']
    if next_cursor is not None:
        if not isinstance(next_cursor, str):
            raise ValueError("'next_cursor' must be null or a string")
        if next_cursor == "":
            raise ValueError("'next_cursor' cannot be an empty string")


def _validate_item(item: Dict[str, Any]) -> None:
    """Validate that an item has the required structure."""
    if not isinstance(item, dict):
        raise ValueError("Item must be a dictionary")
    
    if 'id' not in item:
        raise ValueError("Item must have 'id' key")
    
    if not isinstance(item['id'], str) or item['id'] == "":
        raise ValueError("'id' must be a nonempty string")
    
    if 'name' not in item:
        raise ValueError("Item must have 'name' key")
    
    if not isinstance(item['name'], str) or item['name'] == "":
        raise ValueError("'name' must be a nonempty string")
    
    if 'updated_at' not in item:
        raise ValueError("Item must have 'updated_at' key")
    
    if not isinstance(item['updated_at'], str) or item['updated_at'] == "":
        raise ValueError("'updated_at' must be a nonempty string")
    
    # Validate ISO 8601 timestamp with timezone
    # Must end with timezone indicator (Z, +HH:MM, or -HH:MM)
    updated_at = item['updated_at']
    if updated_at.endswith('Z'):
        pass  # Valid UTC
    elif len(updated_at) >= 6 and updated_at[-6] in ('+', '-') and updated_at[-3] == ':':
        # Check if the last 5 chars after the sign are digits and colon
        offset_part = updated_at[-6:]
        if offset_part[1:3].isdigit() and offset_part[4:6].isdigit():
            pass  # Valid offset like +00:00 or -05:00
        else:
            raise ValueError("'updated_at' must include timezone")
    else:
        raise ValueError("'updated_at' must include timezone")


def _validate_sync_params(max_attempts: int, base_delay: float) -> None:
    """Validate sync parameters before fetching."""
    # max_attempts must be a positive integer (not boolean)
    if isinstance(max_attempts, bool):
        raise ValueError("max_attempts must be an integer, not a boolean")
    if not isinstance(max_attempts, int) or max_attempts <= 0:
        raise ValueError("max_attempts must be a positive integer")
    
    # base_delay must be a nonnegative finite number
    if isinstance(base_delay, bool):
        raise ValueError("base_delay must be a number, not a boolean")
    if not isinstance(base_delay, (int, float)):
        raise ValueError("base_delay must be a number")
    if base_delay < 0:
        raise ValueError("base_delay must be nonnegative")
    if not (base_delay == base_delay):  # Check for NaN
        raise ValueError("base_delay must be finite")


def sync(connection: sqlite3.Connection, fetch_page: Callable[[Optional[str]], Dict[str, Any]], 
         sleep: Callable[[float], None], *, max_attempts: int = 3, base_delay: float = 0.1, 
         logger: Optional[logging.Logger] = None) -> int:
    """
    Synchronize data from an API to SQLite.
    
    Args:
        connection: SQLite connection
        fetch_page: Function that fetches a page given a cursor (None for first page)
        sleep: Function to sleep for a given number of seconds
        max_attempts: Maximum retry attempts per page
        base_delay: Base delay for exponential backoff
        logger: Optional logger for successful completion
    
    Returns:
        Number of unique records committed
    """
    # Validate parameters before any fetching
    _validate_sync_params(max_attempts, base_delay)
    
    # Check if there's an active transaction
    if connection.in_transaction:
        raise ValueError("Cannot sync with an active transaction")
    
    # Collect all pages first, validating them all before any database changes
    all_items: Dict[str, Dict[str, str]] = {}  # id -> {id, name, updated_at}
    cursor: Optional[str] = None
    seen_cursors: set = set()
    
    while True:
        # Validate cursor cycle detection
        if cursor is not None:
            if cursor in seen_cursors:
                raise ValueError("Repeated cursor detected")
            seen_cursors.add(cursor)
        
        # Fetch page with retry logic
        page = None
        for attempt in range(1, max_attempts + 1):
            try:
                page = fetch_page(cursor)
                break
            except TransientError:
                if attempt < max_attempts:
                    sleep(base_delay * (2 ** (attempt - 1)))
                else:
                    raise
        
        # Validate page structure
        _validate_page(page)
        
        # Validate and collect items
        for item in page['items']:
            _validate_item(item)
            all_items[item['id']] = {
                'id': item['id'],
                'name': item['name'],
                'updated_at': item['updated_at']
            }
        
        # Check if there are more pages
        cursor = page['next_cursor']
        if cursor is None:
            break
    
    # All pages validated, now perform database transaction
    # Create table if it doesn't exist
    connection.execute("""
        CREATE TABLE IF NOT EXISTS records (
            id TEXT PRIMARY KEY,
            name TEXT NOT NULL,
            updated_at TEXT NOT NULL
        )
    """)
    
    # Upsert all items
    for item_data in all_items.values():
        connection.execute("""
            INSERT INTO records (id, name, updated_at)
            VALUES (?, ?, ?)
            ON CONFLICT(id) DO UPDATE SET
                name = excluded.name,
                updated_at = excluded.updated_at
        """, (item_data['id'], item_data['name'], item_data['updated_at']))
    
    # Commit the transaction
    connection.commit()
    
    # Log successful completion
    if logger is not None:
        logger.info("Sync completed successfully")
    
    return len(all_items)


def main(argv: Optional[List[str]] = None) -> int:
    """
    CLI entry point for the sync application.
    
    Args:
        argv: Command line arguments (defaults to sys.argv[1:])
    
    Returns:
        Exit code: 0 for success, 1 for sync failure, 2 for invalid configuration
    """
    parser = argparse.ArgumentParser(description='Sync API data to SQLite')
    parser.add_argument('--url', help='API URL (overrides SYNC_URL env var)')
    parser.add_argument('--db', help='SQLite database path (overrides SYNC_DB env var)')
    parser.add_argument('--max-attempts', type=int, help='Max retry attempts (overrides SYNC_MAX_ATTEMPTS)')
    parser.add_argument('--base-delay', type=float, help='Base delay for backoff (overrides SYNC_BASE_DELAY)')
    
    try:
        args = parser.parse_args(argv)
    except SystemExit as e:
        return 2 if e.code != 0 else 0
    
    # Get configuration from args or environment variables
    url = args.url or os.environ.get('SYNC_URL')
    db_path = args.db or os.environ.get('SYNC_DB')
    max_attempts = args.max_attempts if args.max_attempts is not None else int(os.environ.get('SYNC_MAX_ATTEMPTS', 3))
    base_delay = args.base_delay if args.base_delay is not None else float(os.environ.get('SYNC_BASE_DELAY', 0.1))
    
    # Validate required configuration
    if not url or not db_path:
        print("Error: --url and --db are required", file=sys.stderr)
        return 2
    
    # Validate retry configuration
    try:
        _validate_sync_params(max_attempts, base_delay)
    except ValueError as e:
        print(f"Error: {e}", file=sys.stderr)
        return 2
    
    # Setup logging
    logger = logging.getLogger('sync_app')
    logger.setLevel(logging.INFO)
    
    # Open database connection
    connection = sqlite3.connect(db_path)
    
    try:
        # Perform sync
        result = sync(connection, lambda c: fetch_http_page(url, c), time.sleep, 
                     max_attempts=max_attempts, base_delay=base_delay, logger=logger)
        return 0
    except Exception as e:
        print(f"Error: {e}", file=sys.stderr)
        return 1
    finally:
        connection.close()


if __name__ == '__main__':
    sys.exit(main())
