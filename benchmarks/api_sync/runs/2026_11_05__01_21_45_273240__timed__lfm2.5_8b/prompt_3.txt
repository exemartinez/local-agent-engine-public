# API-to-SQLite synchronization specification

## Contract

Implement `sync_app.py` using Python's standard library. Expose `sync(connection, fetch_page, sleep, *, max_attempts=3, base_delay=0.1, logger=None) -> int`, `fetch_http_page(url, cursor)`, and `main(argv=None) -> int`. The model may add helpers but must preserve these signatures. `connection` is a `sqlite3.Connection`; `fetch_page(cursor)` returns a decoded page dictionary. `sleep(seconds)` is injectable. The sync result is the number of unique records committed.

Each page has exactly the required keys `items` (list) and `next_cursor` (null or a nonempty string). An item is a dictionary with `id` (nonempty string), `name` (nonempty string), and `updated_at` (nonempty ISO 8601 timestamp including a timezone). Preserve the timestamp text. Extra page or item keys are permitted and ignored. Duplicate IDs across pages are resolved by the last occurrence in fetch order.

Fetch the first page using cursor `None`, then follow `next_cursor` until null. Detect a repeated cursor and raise `ValueError`. Invalid pages or records raise `ValueError`. All pages must be validated before any persistent database change. Upsert into `records(id TEXT PRIMARY KEY, name TEXT NOT NULL, updated_at TEXT NOT NULL)` in one transaction; repeated runs must not duplicate rows. A failed fetch, validation, or database operation leaves the database as it was before the call, including if the table did not previously exist. Do not commit or roll back a transaction already owned by the caller: reject an already-active transaction with `ValueError` before fetching.

Define `TransientError` for retryable fetch failures. Retry each individual page at most `max_attempts` total calls, sleeping `base_delay * 2**(attempt-1)` after each non-final `TransientError`. Other exceptions propagate immediately. Validate `max_attempts` is a positive integer excluding booleans and `base_delay` is a nonnegative finite number before fetching. Log successful completion without logging credentials or record contents.

`fetch_http_page` performs a GET using `urllib.request`, appending a URL-encoded `cursor` query parameter only when cursor is not null, preserving existing query parameters. It decodes UTF-8 JSON. HTTP 429, HTTP 5xx, and `urllib.error.URLError` raise `TransientError`; other HTTP errors propagate. No network access is needed in tests.

CLI: `--url` and `--db` override `SYNC_URL` and `SYNC_DB` environment variables; both are required. `--max-attempts` and `--base-delay` override `SYNC_MAX_ATTEMPTS` and `SYNC_BASE_DELAY`, defaults 3 and 0.1. `main` returns 0 on success, 2 for invalid configuration, 1 for sync failure, and writes useful diagnostics to stderr. Running the file directly exits with that code. Do not log the URL because it may contain credentials.

"""Synchronize a SQLite database with paginated HTTP pages."""
import sqlite3
import json
import urllib.request
import urllib.error
import time
import os
import sys
from datetime import datetime
from typing import Dict, Any, Optional, Callable


class TransientError(Exception):
    """Raised for retryable fetch failures."""
    pass


def _validate_config(max_attempts: int, base_delay: float) -> None:
    """Validate configuration parameters."""
    if not isinstance(max_attempts, int) or max_attempts <= 0:
        raise ValueError("max_attempts must be a positive integer")
    if not (isinstance(base_delay, (int, float)) and base_delay >= 0):
        raise ValueError("base_delay must be a nonnegative finite number")


def _parse_url_params(url: str) -> Dict[str, Any]:
    """Parse URL query parameters preserving existing ones."""
    parsed = urllib.parse.urlparse(url)
    if parsed.query:
        return dict(urllib.parse.parse_qs(parsed.query))
    return {}


def fetch_http_page(url: str, cursor: Optional[str]) -> Dict[str, Any]:
    """
    Perform a GET request to `url`, appending a URL‑encoded ``cursor`` query
    parameter when ``cursor`` is not ``None``. Returns the decoded JSON payload.
    Raises : TransientError for HTTP 429, 5xx errors and URLError; otherwise
    propagates other exceptions.
    """
    if cursor is None:
        return _parse_url_params(url)

    params = _parse_url_params(url)
    if params:
        # Replace existing cursor param with the new value
        params['cursor'] = urllib.parse.urlencode({'cursor': cursor})
    full_url = url + '?' + params.get('cursor', '')
    try:
        with urllib.request.urlopen(full_url) as response:
            return json.load(response)
    except urllib.error.HTTPError as e:
        if e.code in (429, 500, 501, 502, 503, 504):
            raise TransientError(e)
        else:
            raise
    except urllib.error.URLError as e:
        raise TransientError(e)


def _validate_item(item: Dict[str, Any]) -> None:
    """Validate that an item has required fields and types."""
    if not isinstance(item, dict):
        raise ValueError("item must be a dictionary")
    required = {'id', 'name', 'updated_at'}
    if set(item.keys()) != required:
        raise ValueError(f"item keys must be {required}, got {set(item.keys())}")
    if not item['id'] or not isinstance(item['id'], str) or not item['id'].strip():
        raise ValueError("item.id must be a non‑empty string")
    if not item['name'] or not isinstance(item['name'], str) or not item['name'].strip():
        raise ValueError("item.name must be a non‑empty string")
    if not item['updated_at']:
        raise ValueError("item.updated_at must be a non‑empty ISO 8601 timestamp")


def sync(
    connection: sqlite3.Connection,
    fetch_page: Callable[[str, Optional[str]], Any],
    sleep: Callable[int, None],
    *,
    max_attempts: int = 3,
    base_delay: float = 0.1,
    logger: Optional[Any] = None,
) -> int:
    """
    Sync a SQLite database with paginated HTTP pages.

    Returns the number of unique records that were committed.
    Raises ValueError for configuration errors, duplicate cursors,
    validation failures, or already‑active transactions.
    """
    _validate_config(max_attempts, base_delay)

    if connection.is_autocommit:
        raise ValueError("already active transaction")

    url = None
    next_cursor = None
    seen_cursors = set()
    unique_ids = set()

    def _commit():
        with connection.cursor() as cur:
            cur.execute('BEGIN')
            try:
                return True
            except sqlite3.IntegrityError:
                cur.execute('ROLLBACK')
                raise

    attempts = 0
    while True:
        if next_cursor is not None:
            page = fetch_page(url, next_cursor)
            next_cursor = page.get('next_cursor')
        else:
            page = fetch_http_page(url, None)

        # Validate page structure
        if not isinstance(page, dict):
            raise ValueError("page must be a dictionary")
        _validate_page(page)
        url = page.get('next_cursor')

        # Detect duplicate cursor
        if next_cursor in seen_cursors:
            raise ValueError(f"duplicate cursor {next_cursor}")

        seen_cursors.add(next_cursor)
        attempts += 1

        if attempts > max_attempts:
            raise ValueError(f"max attempts ({max_attempts}) exceeded")

        # Apply delays for transient errors
        if attempts < max_attempts:
            delay = base_delay * (2 ** (attempts - 1))
            sleep(delay)

    # All pages validated – upsert into records table
    try:
        with connection.cursor() as cur:
            cur.execute('BEGIN')
            for item in unique_ids:
                cur.execute(
                    'INSERT OR REPLACE INTO records (id, name, updated_at) VALUES (?, ?, ?)',
                    (item['id'], item['name'], item['updated_at'])
                )
            commit = _commit()
            if commit:
                return len(unique_ids)
    except Exception:
        # Rollback any partial changes
        try:
            cur.execute('ROLLBACK')
        except sqlite3.Error:
            pass
        raise


def _validate_page(page: Dict[str, Any]) -> None:
    """Validate a page's pagination and item data."""
    if not isinstance(page, dict):
        raise ValueError("page must be a dictionary")
    required_keys = {'items', 'next_cursor'}
    if set(page.keys()) != required_keys:
        raise ValueError(f"page keys must be {required_keys}, got {set(page.keys())}")
    items = page['items']
    if not isinstance(items, list):
        raise ValueError("items must be a list")
    for item in items:
        _validate_item(item)


def main(argv: Optional[list]) -> int:
    """CLI entry point."""
    url = os.getenv('SYNC_URL')
    db_path = os.getenv('SYNC_DB')

    if not url or not db_path:
        sys.stderr.write("ERROR: SYNC_URL and SYNC_DB environment variables must be set\n")
        return 2

    try:
        sync(
            sqlite3.connect(db_path),
            fetch_http_page,
            sleep,
            max_attempts=argv[1] if len(argv) > 1 else 3,
            base_delay=argv[2] if len(argv) > 2 else 0.1,
        )
    except ValueError:
        sys.stderr.write("ERROR: invalid configuration\n")
        return 2
    except Exception:
        sys.stderr.write("ERROR: sync failure\n")
        return 1

    return 0
